import { Meta } from '@storybook/addon-docs/blocks'

import { tokens } from '../../tokens/mantis.tokens.mjs'
import { resolveValue } from '../../scripts/token-model.mjs'

<Meta title="Foundations / Tokens" />

export const modes = tokens.modes

{/* Roles may alias a source token ('{color-lime-500}'); resolve it for the swatch. */}
export const resolved = (value, mode) => resolveValue(tokens, value, mode)

export const Chip = ({ value }) => (
  <span
    aria-hidden="true"
    style={{
      display: 'inline-block',
      width: 14,
      height: 14,
      marginRight: 8,
      verticalAlign: 'middle',
      borderRadius: 3,
      boxShadow: 'inset 0 0 0 1px rgba(0, 0, 0, 0.2)',
      background: value,
    }}
  />
)

# Tokens

Every token in `tokens/mantis.tokens.mjs`, the single source that the build scripts turn into the theme CSS, the DTCG export and this Storybook's own colors. System <code>{tokens.meta.name}</code>, version <code>{tokens.meta.version}</code>.

## Color

<table>
  <thead>
    <tr>
      <th>Token</th>
      <th>Variable</th>
      {modes.map((mode) => <th key={mode}>{mode}</th>)}
    </tr>
  </thead>
  <tbody>
    {Object.entries(tokens.color).flatMap(([group, roles]) =>
      Object.entries(roles).map(([role, value]) => (
        <tr key={`${group}.${role}`}>
          <td><code>{`color.${group}.${role}`}</code></td>
          <td><code>{`--${group}-${role}`}</code></td>
          {modes.map((mode) => (
            <td key={mode}><Chip value={resolved(value[mode], mode)} /><code>{value[mode]}</code></td>
          ))}
        </tr>
      )),
    )}
  </tbody>
</table>
